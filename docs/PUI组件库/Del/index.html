PuiDel ​

  • 全局通用的删除

WARNING

已弃用, 使用框架项目组件方式维护

使用 ​

js
import PuiDel from '@Pui/Del'

/** render 函数中使用 删除按钮  */
 h(PuiDel,{
    props:{
      delApi:()=>{}, //删除接口 axios对象
      delParams:{}, //需要的数据
    },
    on: {
      onSuccess: ()=>{ console.log(params) } //成功的回调
    }
  }),
html
/** 模板中使用 */
<PuiDel :delApi='' :delParams='' @onSuccess='' />
